Quick Start

curl -X POST https://web2md.onrender.com/convert \ -H "Content-Type: application/json" \ -d '{"url": "https://example.com/article"}'
import httpx resp = httpx.post("https://web2md.onrender.com/convert", json={ "url": "https://example.com/article" }) data = resp.json() print(data["title"]) print(data["content"])
const resp = await fetch("https://web2md.onrender.com/convert", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ url: "https://example.com/article" }) }); const data = await resp.json(); console.log(data.title, data.content);
